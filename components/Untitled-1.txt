"use client";

import { useState } from "react";
import { FaGoogle, FaApple } from "react-icons/fa";
import { Sparkles } from "lucide-react";
import { motion, AnimatePresence } from "framer-motion";

// GlowButton : bouton premium bleu brillant
function GlowButton({ children, className = "", loading = false, ...props }) {
  return (
    <motion.button
      whileHover={{ scale: 1.04, boxShadow: "0 0 24px #3b82f6, 0 0 8px #fff" }}
      whileTap={{ scale: 0.97 }}
      className={`relative overflow-hidden px-6 py-3 rounded-xl font-semibold bg-gradient-to-br from-blue-500 to-blue-700 text-white shadow-lg border border-blue-400/40 focus:outline-none focus:ring-2 focus:ring-blue-400 transition-all duration-200 ${className}`}
      disabled={loading}
      {...props}
    >
      <span className="absolute inset-0 pointer-events-none z-0 animate-glow" />
      <span className="relative z-10 flex items-center gap-2">
        {loading && <AeternumLoader size={20} />}
        {children}
      </span>
      <style jsx>{`
        .animate-glow {
          box-shadow: 0 0 32px 8px #3b82f6, 0 0 8px #fff;
          opacity: 0.5;
          border-radius: 1rem;
          animation: glow 2s infinite alternate;
        }
        @keyframes glow {
          0% { opacity: 0.5; box-shadow: 0 0 16px 4px #3b82f6, 0 0 4px #fff; }
          100% { opacity: 1; box-shadow: 0 0 40px 16px #60a5fa, 0 0 16px #fff; }
        }
      `}</style>
    </motion.button>
  );
}

// Loader Aeternum bleu
function AeternumLoader({ size = 32 }) {
  return (
    <span className="inline-block align-middle">
      <span
        className="block border-4 border-blue-400 border-t-transparent rounded-full animate-spin"
        style={{ width: size, height: size }}
      />
    </span>
  );
}

// Avatar animé Malaika
function MalaikaAvatar() {
  return (
    <motion.div
      initial={{ scale: 0, rotate: -30 }}
      animate={{ scale: 1, rotate: 0 }}
      transition={{ type: "spring", stiffness: 200, damping: 12 }}
      className="mx-auto mb-2 w-16 h-16 rounded-full bg-gradient-to-br from-blue-400 to-blue-700 border-4 border-white shadow-lg flex items-center justify-center overflow-hidden"
    >
      <motion.img
        src="/malaika-avatar.png"
        alt="Malaika"
        className="w-12 h-12 object-cover"
        initial={{ y: -20, opacity: 0 }}
        animate={{ y: 0, opacity: 1 }}
        transition={{ delay: 0.2 }}
      />
    </motion.div>
  );
}

// Boutons variantes
function GoogleButton(props) {
  return (
    <GlowButton {...props} className="w-full flex items-center gap-2 justify-center bg-white text-blue-700 border border-blue-200 hover:bg-blue-50">
      <FaGoogle className="text-lg" />
      <span>Connexion Google</span>
    </GlowButton>
  );
}
function AppleButton(props) {
  return (
    <GlowButton {...props} className="w-full flex items-center gap-2 justify-center bg-black text-white border border-gray-700 hover:bg-gray-900">
      <FaApple className="text-lg" />
      <span>Connexion Apple</span>
    </GlowButton>
  );
}
function MagicLinkButton(props) {
  return (
    <GlowButton {...props} className="w-full flex items-center gap-2 justify-center bg-gradient-to-br from-blue-400 to-blue-700 text-white">
      <Sparkles className="text-lg" />
      <span>Connexion magique</span>
    </GlowButton>
  );
}
function PrimaryButton({ loading, ...props }) {
  return (
    <GlowButton {...props} loading={loading} className="w-full">
      {props.children}
    </GlowButton>
  );
}

// Page principale
export default function StaffAeternumLogin() {
  const [loading, setLoading] = useState(false);
  const [showForm, setShowForm] = useState(true);

  return (
    <div className="min-h-screen flex items-center justify-center px-2 py-8 bg-gradient-to-b from-slate-100 to-blue-100 dark:from-slate-900 dark:to-black text-slate-900 dark:text-white">
      <motion.div
        initial={{ opacity: 0, y: 40 }}
        animate={{ opacity: 1, y: 0 }}
        transition={{ duration: 0.7, ease: "easeOut" }}
        className="w-full max-w-md space-y-6 p-6 rounded-2xl bg-white/80 dark:bg-black/60 backdrop-blur-xl border border-white/20 shadow-2xl"
      >
        <MalaikaAvatar />
        <h1 className="text-2xl font-bold text-center tracking-tight mb-2">
          Aeternum — Espace Staff
        </h1>
        <p className="text-center text-sm text-blue-700 dark:text-blue-200 mb-4">
          Connecte-toi pour rejoindre le Staff Hub.
        </p>
        <AnimatePresence>
          {showForm && (
            <motion.div
              initial={{ opacity: 0, y: 20 }}
              animate={{ opacity: 1, y: 0 }}
              exit={{ opacity: 0, y: -20 }}
              transition={{ duration: 0.5 }}
              className="space-y-3"
            >
              <GoogleButton onClick={() => setLoading(true)} loading={loading} />
              <AppleButton onClick={() => setLoading(true)} loading={loading} />
              <MagicLinkButton onClick={() => setLoading(true)} loading={loading} />
            </motion.div>
          )}
        </AnimatePresence>
        <div className="flex items-center gap-3 my-2">
          <div className="h-px flex-1 bg-blue-200 dark:bg-white/20"></div>
          <span className="text-xs text-blue-400 dark:text-white/50">ou</span>
          <div className="h-px flex-1 bg-blue-200 dark:bg-white/20"></div>
        </div>
        <form
          className="space-y-4"
          onSubmit={e => {
            e.preventDefault();
            setLoading(true);
            setTimeout(() => setLoading(false), 1800);
          }}
        >
          <div>
            <label className="text-sm">Email</label>
            <input
              type="email"
              required
              className="w-full mt-1 px-3 py-2 rounded-lg bg-white/60 dark:bg-white/10 border border-blue-200 dark:border-white/20 focus:outline-none focus:ring-2 focus:ring-blue-400 text-slate-900 dark:text-white"
            />
          </div>
          <div>
            <label className="text-sm">Mot de passe</label>
            <input
              type="password"
              required
              className="w-full mt-1 px-3 py-2 rounded-lg bg-white/60 dark:bg-white/10 border border-blue-200 dark:border-white/20 focus:outline-none focus:ring-2 focus:ring-blue-400 text-slate-900 dark:text-white"
            />
          </div>
          <PrimaryButton type="submit" loading={loading}>
            Connexion
          </PrimaryButton>
          <PrimaryButton type="button" onClick={() => setShowForm(false)}>
            Inscription
          </PrimaryButton>
        </form>
      </motion.div>
    </div>
  );
}
